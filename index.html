function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var payload = JSON.parse(e.postData.contents); if (payload.action === "batchReplace") { var newRows = payload.rows; var headers = [ "Plate", "Brand", "Model", "VIN", "Fuel Type", "Total KM", "Availability", "Accidents", "Last Maintenance", "Last Oil Change", "Insurance Start Date", "Insurance Expiry Date", "Operation Card Start Date", "Operation Card Expiry Date" ]; var lastRow = sheet.getLastRow(); if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, headers.length).clearContent(); if (newRows && newRows.length > 0) { var outputData = []; for (var i = 0; i < newRows.length; i++) { var rowObj = newRows[i]; var rowArray = []; for (var j = 0; j < headers.length; j++) { var val = rowObj[headers[j]]; rowArray.push(val !== undefined && val !== null ? val : ""); } outputData.push(rowArray); } sheet.getRange(2, 1, outputData.length, headers.length).setValues(outputData); } return ContentService.createTextOutput(JSON.stringify({ status: "success", count: newRows ? newRows.length : 0 })).setMimeType(ContentService.MimeType.JSON); } var data = sheet.getDataRange().getValues(); var headers = data[0]; var plateIndex = headers.indexOf("Plate"); var targetRowIndex = -1; for (var i = 1; i < data.length; i++) { if (data[i][plateIndex].toString().trim() === payload["Plate"].toString().trim()) { targetRowIndex = i + 1; break; } } if (targetRowIndex !== -1) { for (var j = 0; j < headers.length; j++) { var headerName = headers[j].toString().trim(); if (payload.hasOwnProperty(headerName)) { sheet.getRange(targetRowIndex, j + 1).setValue(payload[headerName]); } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Vehicle not found" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
